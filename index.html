function scrollToAnchor(hash) { if (!hash || hash === '#') return; const target = document.querySelector(hash); if (!target) return; const yOffset = -20; const y = target.getBoundingClientRect().top + window.pageYOffset + yOffset; window.scrollTo({ top: y, behavior: 'smooth' }); } window.addEventListener('load', function() { setTimeout(function() { scrollToAnchor(window.location.hash); }, 500); }); document.addEventListener('click', function(e) { const link = e.target.closest('a[href^="#"]'); if (!link) return; const href = link.getAttribute('href'); if (!href || href === '#') return; const target = document.querySelector(href); if (!target) return; e.preventDefault(); scrollToAnchor(href); history.pushState(null, null, href); });
